local function newSlider(parent, label, min, max, getter, setter)
    local holder = create("Frame", {
        Size = UDim2.new(1, -8, 0, 56 * UI_SCALE),
        BackgroundColor3 = Color3.fromRGB(18, 20, 25),
        BorderSizePixel = 0,
    }, parent)
    create("UICorner", { CornerRadius = UDim.new(0, 10) }, holder)

    local lbl = create("TextLabel", {
        Size = UDim2.new(1, -16, 0, 20 * UI_SCALE),
        Position = UDim2.new(0, 12, 0, 4),
        BackgroundTransparency = 1,
        Text = label .. ": " .. getter(),
        TextColor3 = Color3.fromRGB(225, 225, 230),
        Font = Enum.Font.Gotham, TextSize = 12 * UI_SCALE,
        TextXAlignment = Enum.TextXAlignment.Left,
    }, holder)

    local track = create("TextButton", {
        Size = UDim2.new(1, -24, 0, 14 * UI_SCALE),
        Position = UDim2.new(0, 12, 0, 30 * UI_SCALE),
        BackgroundColor3 = Color3.fromRGB(32, 34, 40),
        Text = "", BorderSizePixel = 0, AutoButtonColor = false,
    }, holder)
    create("UICorner", { CornerRadius = UDim.new(0, 8) }, track)
    local fill = create("Frame", {
        Size = UDim2.new((getter() - min) / (max - min), 0, 1, 0),
        BackgroundColor3 = COL, BorderSizePixel = 0,
    }, track)
    create("UICorner", { CornerRadius = UDim.new(0, 8) }, fill)

    local dragging = false
    local function update(input)
        local rel = math.clamp(
            (input.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
        local val = math.floor(min + (max - min) * rel)
        fill.Size = UDim2.new(rel, 0, 1, 0)
        lbl.Text = label .. ": " .. val
        setter(val)
    end
    track.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch
        or input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true update(input)
        end
    end)
    track.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.Touch
        or input.UserInputType == Enum.UserInputType.MouseMovement) then
            update(input)
        end
    end)
    track.InputEnded:Connect(function() dragging = false end)
end

-- ═══════════════════════════════════════════════════════════
-- KILLAURA — один режим: орбита вокруг цели
-- ═══════════════════════════════════════════════════════════

local function getClosestTarget()
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local closest, closestDist = nil, math.huge
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and isSpawned(plr) then
            local thrp = plr.Character:FindFirstChild("HumanoidRootPart")
            if thrp then
                local d = (hrp.Position - thrp.Position).Magnitude
                if d < closestDist then
                    closest, closestDist = plr, d
                end
            end
        end
    end
    return closest
end

local function activateTool()
    local char = LocalPlayer.Character
    if not char then return end
    local tool = char:FindFirstChildOfClass("Tool")
    if tool then
        pcall(function() tool:Activate() end)
    end
end

local orbitAngle = 0
local kaAccum = 0

RunService.RenderStepped:Connect(function(dt)
    if not Config.KillAura then return end

    local target = getClosestTarget()
    if not target then return end

    local tChar = target.Character
    local tHrp = tChar and tChar:FindFirstChild("HumanoidRootPart")
    if not tHrp then return end

    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    local dist = (hrp.Position - tHrp.Position).Magnitude
    if dist > Config.KillAuraRange then return end

    -- Орбитальное движение вокруг цели
    orbitAngle = orbitAngle + dt * Config.OrbitSpeed
    local offset = Vector3.new(
        math.cos(orbitAngle) * (Config.KillAuraRange * 0.5),
        0,
        math.sin(orbitAngle) * (Config.KillAuraRange * 0.5)
    )
    pcall(function()
        hrp.CFrame = CFrame.new(tHrp.Position + offset, tHrp.Position)
    end)

    -- Атака
    kaAccum = kaAccum + dt
    if kaAccum >= Config.KillAuraSpeed then
        kaAccum = 0
        activateTool()
    end
end)

local flyBV, flyBG

        return
    end
    local char = LocalPlayer.Character

        flyBV = Instance.new("BodyVelocity")

    end
    if not flyBG then
        flyBG = Instance.new("BodyGyro")
        flyBG.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
        flyBG.P = 9e4
        flyBG.Parent = hrp
    end
    flyBG.CFrame = Camera.CFrame
    if IS_MOBILE then
        flyBV.Velocity = Camera.CFrame.LookVector * Config.FlySpeed
    else
        local mv = Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then mv += Camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then mv -= Camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then mv -= Camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then mv += Camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then mv += Vector3.new(0, 1, 0) end
        if mv.Magnitude > 0 then
            flyBV.Velocity = mv.Unit * Config.FlySpeed
        else
            flyBV.Velocity = Vector3.zero
        end
    end
end)

    local char = LocalPlayer.Character
    if not char then return end
    if Config.Noclip then
        for _, p in ipairs(char:GetDescendants()) do
            if p:IsA("BasePart") and p.CanCollide then
                p.CanCollide = false
            end
        end
    else
        for _, p in ipairs(char:GetDescendants()) do
            if p:IsA("BasePart") and not p.CanCollide then
                p.CanCollide = true
            end
        end
    end
end)

    if not hum then return end
    if Config.Speed then

        end
    end
end)

-- Spin
local spinGyro, spinAngle = nil, 0
RunService.Heartbeat:Connect(function(dt)
    local char = LocalPlayer.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    if Config.Spin then
        if not spinGyro or spinGyro.Parent ~= hrp then
            if spinGyro then pcall(function() spinGyro:Destroy() end) end
            spinGyro = Instance.new("BodyGyro")
            spinGyro.MaxTorque = Vector3.new(0, math.huge, 0)
            spinGyro.P = 50000
            spinGyro.D = 500
            spinGyro.Parent = hrp
        end
        spinAngle = spinAngle + dt * Config.SpinSpeed * 5
        spinGyro.CFrame = CFrame.new(hrp.Position) * CFrame.Angles(0, spinAngle, 0)
    elseif spinGyro then
        pcall(function() spinGyro:Destroy() end)
        spinGyro = nil
        spinAngle = 0
    end
end)
